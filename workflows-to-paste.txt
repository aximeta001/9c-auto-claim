name: 9C 自动领奖 (Heimdall)

# ═══════════════════════════════════════════════════════════════════════════
# 架构：纯 GitHub 自接力长链，完全不依赖 WorkBuddy / 任何外部闹钟
# ───────────────────────────────────────────────────────────────────────────
#  点火：手动 Run workflow 一次（取消勾选 dry_run），或等 schedule 偶尔命中。
#  值守：job 内 `claim.mjs --daemon` 常驻 5.5h，按链上区块休眠/轮询，
#        谁到期就领谁，跨越多个奖励周期，领完不退出。
#        同一条链内串行值守 3 个角色（bnb / bnbyes / tomoon）：三者同属一个
#        agent、共用同一私钥与 nonce 队列，必须串行提交，绝不开并行链抢 nonce。
#        角色列表默认内置在脚本里，也可用仓库 Variable `NC_AVATARS` 覆盖。
#  接力：预算耗尽后脚本正常退出，最后一个 step 用 workflow_dispatch 触发
#        下一个常驻 job。官方文档明确 workflow_dispatch / repository_dispatch
#        是「GITHUB_TOKEN 防递归规则」的例外，会创建新 workflow run。
#        concurrency 同组只保留 1 running + 1 pending → 无缝串行交接。
#  重燃：schedule 每 10 分钟错峰，只作「接力链意外断裂时重新点火」的保险。
#        GitHub schedule 高负载会延迟/丢弃，不可靠，但链正常时不需要它。
#
#  关键：领取精度由 job 内部的区块轮询保证（±1 分钟），与闹钟准时性无关。
#  每次触发都是全新 runner：代码由 checkout 提供，密钥由 Secrets 注入，
#  私钥只在 runner 内由加密 keystore 临时解密成 .env，job 结束即销毁。
# ═══════════════════════════════════════════════════════════════════════════
on:
  schedule:
    # 重燃种子：刻意避开 :00 整点高峰。命中率低不影响——
    # 它只需在接力链断掉后偶尔成功一次，把链重新点起来。
    - cron: '4,14,24,34,44,54 * * * *'
  workflow_dispatch:
    inputs:
      dry_run:
        description: '手动验证时勾选：逐一核对 3 个角色链上状态，不值守、不提交、不接力'
        type: boolean
        default: true
      max_run_seconds:
        description: '仅用于实测接力链：常驻预算秒数，留空=默认 5.5h；填 60 可在一分钟内验证交棒'
        type: string
        default: ''

permissions:
  contents: read
  actions: write      # 接力 step 调用 workflow_dispatch REST API 必需

concurrency:
  group: 9c-auto-claim
  cancel-in-progress: false   # 绝不能取消正在值守的 job
  # 同组只保留 1 个 running + 1 个 pending。接力 dispatch 的新 run 排队，
  # 当前 job 一结束就接上；schedule 触发的也进同一组排队，永不并发抢 nonce。

jobs:
  claim:
    runs-on: ubuntu-latest
    # 单 job 上限 6h(360min)。脚本预算 330min，留 25min 给安装、收尾与接力 step，
    # 避免被 timeout 强杀导致接力 step 来不及执行。
    timeout-minutes: 355
    steps:
      - name: 检出代码
        uses: actions/checkout@v4

      - name: 准备 Node
        uses: actions/setup-node@v4
        with:
          node-version: '22'
          cache: npm

      - name: 安装依赖（命中缓存时仅数秒）
        run: npm ci --omit=dev --no-audit --no-fund

      - name: 解密 keystore 生成 .env（仅本 runner 内，600 权限）
        env:
          NC_KEYSTORE_JSON: ${{ secrets.NC_KEYSTORE_JSON }}
          NC_KEYSTORE_PASS: ${{ secrets.NC_KEYSTORE_PASS }}
          NC_AVATAR: ${{ vars.NC_AVATAR }}
        run: |
          set -e
          if [ -z "$NC_KEYSTORE_JSON" ] || [ -z "$NC_KEYSTORE_PASS" ]; then
            echo "缺少 NC_KEYSTORE_JSON / NC_KEYSTORE_PASS Secrets"; exit 1
          fi
          umask 077
          trap 'rm -f .keystore.json .kpass' EXIT
          printf '%s' "$NC_KEYSTORE_JSON" > .keystore.json
          printf '%s' "$NC_KEYSTORE_PASS" > .kpass
          ARGS="--keystore .keystore.json --passphrase-file .kpass"
          if [ -n "$NC_AVATAR" ]; then ARGS="$ARGS --avatar $NC_AVATAR"; fi
          node import-key.mjs $ARGS

      - name: 常驻值守并领取（--daemon）
        id: daemon
        env:
          DRY_RUN: ${{ inputs.dry_run }}
          EVENT_NAME: ${{ github.event_name }}
          MAX_RUN: ${{ inputs.max_run_seconds }}
          # 可选覆盖：逗号分隔的 "名字:0x地址"。留空即用脚本内置的三角色。
          NC_AVATARS: ${{ vars.NC_AVATARS }}
        run: |
          set -o pipefail
          # 仅调试用：允许把常驻预算临时改短，快速验证交棒；留空则走默认 5.5h。
          EXTRA=""
          case "$MAX_RUN" in
            ''|*[!0-9]*) : ;;                 # 空或非数字 → 不加参数
            *) EXTRA="--max-run-seconds $MAX_RUN" ;;
          esac
          if [ "$EVENT_NAME" = "workflow_dispatch" ] && [ "$DRY_RUN" = "true" ]; then
            echo "手动验证模式：逐一核对 3 个角色链上状态，不值守、不提交、不接力"
            node claim.mjs --check-all 2>&1 | tee "$RUNNER_TEMP/daemon.log"
          else
            # 真实常驻：守满预算，期间按链上区块自等待、到期即领、领完继续。
            # 输出 tee 到文件，供接力 step 校验是否为「正常交棒」。
            node claim.mjs --daemon $EXTRA 2>&1 | tee "$RUNNER_TEMP/daemon.log"
          fi
          exit ${PIPESTATUS[0]}

      - name: 接力（触发下一个常驻 job）
        if: always()
        env:
          # 未配置 PAT 时用内置 GITHUB_TOKEN（零配置即可跑）；
          # 若发现自接力被平台限制，可在网页手动加 Secret NC_DISPATCH_TOKEN 兜底。
          DISPATCH_TOKEN: ${{ secrets.NC_DISPATCH_TOKEN || github.token }}
          RELAY_REPO: ${{ github.repository }}
          RELAY_REF: ${{ github.ref_name }}
          EVENT_NAME: ${{ github.event_name }}
          DRY_RUN: ${{ inputs.dry_run }}
        run: |
          # 只在「预算耗尽、正常交棒」时接力。演练、异常中止、安装失败等
          # 绝不接力，避免高频空转循环刷爆 run 次数；此时由 schedule 重燃。
          if [ "$EVENT_NAME" = "workflow_dispatch" ] && [ "$DRY_RUN" = "true" ]; then
            echo "手动验证模式：不接力"; exit 0
          fi
          LOG="$RUNNER_TEMP/daemon.log"
          if ! grep -q '"handoff":"budget_exhausted"' "$LOG" 2>/dev/null; then
            echo "未检测到正常交棒标记，不接力（由 schedule 兜底重燃）"; exit 0
          fi
          echo "检测到正常交棒，发起接力…"
          CODE=""
          for i in 1 2 3; do
            CODE=$(curl -sS -m 30 -o "$RUNNER_TEMP/relay_resp.txt" -w "%{http_code}" -X POST \
              -H "Authorization: Bearer $DISPATCH_TOKEN" \
              -H "Accept: application/vnd.github+json" \
              -H "X-GitHub-Api-Version: 2022-11-28" \
              -d "{\"ref\":\"$RELAY_REF\",\"inputs\":{\"dry_run\":\"false\"}}" \
              "https://api.github.com/repos/$RELAY_REPO/actions/workflows/claim.yml/dispatches") || CODE="000"
            if [ "$CODE" = "204" ] || [ "$CODE" = "200" ]; then break; fi
            echo "接力第 $i 次返回 HTTP=$CODE，5s 后重试"
            sleep 5
          done
          echo "接力 dispatch HTTP=$CODE"
          if [ "$CODE" != "204" ] && [ "$CODE" != "200" ]; then
            echo "::warning::接力失败 HTTP=$CODE，链将依赖 schedule 重燃"
            head -c 300 "$RUNNER_TEMP/relay_resp.txt" 2>/dev/null || true
          fi
          # 接力失败不算 job 失败：daemon 的领取工作已正常完成。
          exit 0
===== END =====

name: 保活（防止定时任务被自动停用）

# GitHub 官方规则：仓库 60 天没有任何活动，会自动禁用 scheduled workflow。
# 本任务每周打一个「空提交」制造一次仓库活动，让领奖任务持续可用。
# 领奖任务只监听 schedule / workflow_dispatch，不会被这个提交触发。
on:
  schedule:
    - cron: '23 4 * * 1'   # 每周一 04:23 UTC
  workflow_dispatch:

permissions:
  contents: write           # 空提交需要写权限，见 DEPLOY.md 第 4 步

concurrency:
  group: 9c-keepalive
  cancel-in-progress: true

jobs:
  keepalive:
    runs-on: ubuntu-latest
    timeout-minutes: 3
    steps:
      - uses: actions/checkout@v4
      - name: 空提交保活
        run: |
          git config user.name "github-actions[bot]"
          git config user.email "41898282+github-actions[bot]@users.noreply.github.com"
          git commit --allow-empty -m "chore: keep scheduled workflows alive"
          git push origin HEAD:"${GITHUB_REF_NAME}"
===== END =====

name: 9C 无限塔 下一赛季自动冲塔 (Heimdall)

# ═══════════════════════════════════════════════════════════════════════════
# 盯《无限塔》新赛季（默认第 11 期），一旦链上排期上线，立刻自动提交一笔
# infinite_tower_battle 打第 1 层 —— 个人补票时钟锚定在「首次进塔」那一笔，
# 越早打第 1 层越早开始计时。
#
# 【为什么不用 schedule 当主力】GitHub 的 schedule cron 在高负载时会延迟数十分钟
#   甚至整点丢弃，抢首层完全靠不住。本 workflow 复用领奖链已验证的高可用架构：
#
#   主力：job 内 infinite-tower.mjs --watch 长驻，进程内每 20 秒轮询链上一次，
#         赛季一排期上链，最多 20 秒内命中并提交（不等下一条 cron）。
#   接力：单个 job 守满预算（默认 5.5h < 6h 上限）后打印
#         {"handoff":"budget_exhausted"}，本文件末尾的接力 step 用
#         workflow_dispatch REST API 即时刻下一个 job，concurrency 组保证串行无缝。
#   停止：冲塔成功 / 交易定论后打印 {"done":...} / 非交棒结果，接力 step 不接力，
#         赛季内不再空转浪费 runner 额度。
#   保险：schedule 每 10 分钟只作「接力链意外断裂时重新点火」，链正常时根本用不到。
#
#  判定（脚本内只读链上权威状态，全部满足才提交）：
#    · infiniteTowerId >= 11 且 isActive 且 clearedFloor < 1 且 remainingTickets >= 1
#    · 第 1 层通关后永久幂等跳过；交易只认 SUCCESS/FAILURE/INVALID 终态，跨节点复核。
#  前置：新赛季排期由官方 patch 注入链上 —— 未上线前脚本只轮询等待，不乱提交。
#
#  私钥：与领奖/扫货/AB 三条链共用同一份 keystore Secrets，仅在 runner 内临时解密，
#        job 结束即销毁。单笔交易，无 AP 消耗、无 NCG/材料花费（新赛季首笔送 1 票）。
#  通知：全程静默，不推送任何通知，结果只在 Actions 日志里。
# ═══════════════════════════════════════════════════════════════════════════

on:
  schedule:
    # 仅「断链重燃」保险：即使被 GitHub 延迟/丢弃也不影响主力接力链。
    - cron: '7,17,27,37,47,57 * * * *'
  workflow_dispatch:
    inputs:
      dry_run:
        description: '默认勾选：只轮询不提交（手工验证用）。要真跑请取消勾选'
        type: boolean
        default: true
      season:
        description: '目标赛季号（默认 11；留空用默认）'
        type: string
        default: '11'
      max_watch_seconds:
        description: '仅调试：本次蹲守预算秒数，留空=默认 5.5h；填 60 可一分钟验证交棒'
        type: string
        default: ''

permissions:
  contents: read
  actions: write      # 接力 step 调用 workflow_dispatch REST API 必需

concurrency:
  group: 9c-infinite-tower        # 独立组：不与 claim/market/boss-sweep 共用
  cancel-in-progress: false       # 接力的新 run 排队，当前 job 一结束就无缝接上，永不并发

jobs:
  tower:
    runs-on: ubuntu-latest
    timeout-minutes: 360           # > 5.5h 蹲守预算，留足交棒余量（< GitHub 单 job 上限 6h）
    steps:
      - name: 检出代码
        uses: actions/checkout@v4

      - name: 准备 Node
        uses: actions/setup-node@v4
        with:
          node-version: '22'
          cache: npm

      - name: 安装依赖
        run: npm ci --omit=dev --no-audit --no-fund

      - name: 解密 keystore 生成 .env（仅本 runner 内，600 权限）
        env:
          NC_KEYSTORE_JSON: ${{ secrets.NC_KEYSTORE_JSON }}
          NC_KEYSTORE_PASS: ${{ secrets.NC_KEYSTORE_PASS }}
          NC_AVATAR: ${{ vars.NC_AVATAR }}
        run: |
          set -e
          if [ -z "$NC_KEYSTORE_JSON" ] || [ -z "$NC_KEYSTORE_PASS" ]; then
            echo "缺少 NC_KEYSTORE_JSON / NC_KEYSTORE_PASS Secrets"; exit 1
          fi
          umask 077
          trap 'rm -f .keystore.json .kpass' EXIT
          printf '%s' "$NC_KEYSTORE_JSON" > .keystore.json
          printf '%s' "$NC_KEYSTORE_PASS" > .kpass
          ARGS="--keystore .keystore.json --passphrase-file .kpass"
          if [ -n "$NC_AVATAR" ]; then ARGS="$ARGS --avatar $NC_AVATAR"; fi
          node import-key.mjs $ARGS

      - name: 纯逻辑自测（不联网）
        run: node infinite-tower.mjs --self-test

      - name: 蹲守新赛季并冲第 1 层（进程内高频轮询）
        env:
          EVENT: ${{ github.event_name }}
          DRY: ${{ inputs.dry_run }}
          SEASON: ${{ inputs.season }}
          WATCH_SEC: ${{ inputs.max_watch_seconds }}
        run: |
          set -o pipefail
          # 主力真跑：--watch 常驻高频轮询。schedule 触发或手工取消 dry_run 勾选 = 真跑。
          EXTRA="--watch --gap-ms 20000"
          if [ "$EVENT" = "workflow_dispatch" ] && [ "$DRY" = "true" ]; then
            EXTRA="$EXTRA --dry-run"
          fi
          case "$SEASON" in
            ''|*[!0-9]*) EXTRA="$EXTRA --season 11" ;;
            *) EXTRA="$EXTRA --season $SEASON" ;;
          esac
          case "$WATCH_SEC" in
            ''|*[!0-9]*) : ;;
            *) EXTRA="$EXTRA --max-watch-seconds $WATCH_SEC" ;;
          esac
          echo "运行参数: $EXTRA"
          node infinite-tower.mjs $EXTRA 2>&1 | tee "$RUNNER_TEMP/infinite-tower.log"
          exit ${PIPESTATUS[0]}

      - name: 接力（预算耗尽时触发下一个蹲守 job）
        if: always()
        env:
          # 未配置 PAT 时用内置 GITHUB_TOKEN（零配置即可跑）；
          # 若自接力被平台限制，可加 Secret NC_DISPATCH_TOKEN 兜底。
          DISPATCH_TOKEN: ${{ secrets.NC_DISPATCH_TOKEN || github.token }}
          RELAY_REPO: ${{ github.repository }}
          RELAY_REF: ${{ github.ref_name }}
          EVENT_NAME: ${{ github.event_name }}
          DRY_RUN: ${{ inputs.dry_run }}
          SEASON: ${{ inputs.season }}
        run: |
          # 只在「预算耗尽、正常交棒」时接力。演练、冲塔成功、异常中止绝不接力，
          # 避免高频空转刷爆 run 次数；断链时由 schedule 重燃。
          if [ "$EVENT_NAME" = "workflow_dispatch" ] && [ "$DRY_RUN" = "true" ]; then
            echo "手动演练模式：不接力"; exit 0
          fi
          LOG="$RUNNER_TEMP/infinite-tower.log"
          if ! grep -q '"handoff":"budget_exhausted"' "$LOG" 2>/dev/null; then
            echo "未检测到正常交棒标记（可能已冲塔成功/定论/异常），不接力"; exit 0
          fi
          echo "检测到正常交棒，发起接力…"
          SEASON_IN="${SEASON:-11}"
          case "$SEASON_IN" in ''|*[!0-9]*) SEASON_IN="11" ;; esac
          CODE=""
          for i in 1 2 3; do
            CODE=$(curl -sS -m 30 -o "$RUNNER_TEMP/relay_resp.txt" -w "%{http_code}" -X POST \
              -H "Authorization: Bearer $DISPATCH_TOKEN" \
              -H "Accept: application/vnd.github+json" \
              -H "X-GitHub-Api-Version: 2022-11-28" \
              -d "{\"ref\":\"$RELAY_REF\",\"inputs\":{\"dry_run\":\"false\",\"season\":\"$SEASON_IN\"}}" \
              "https://api.github.com/repos/$RELAY_REPO/actions/workflows/infinite-tower.yml/dispatches") || CODE="000"
            if [ "$CODE" = "204" ] || [ "$CODE" = "200" ]; then break; fi
            echo "接力第 $i 次返回 HTTP=$CODE，5s 后重试"
            sleep 5
          done
          echo "接力 dispatch HTTP=$CODE"
          if [ "$CODE" != "204" ] && [ "$CODE" != "200" ]; then
            echo "::warning::接力失败 HTTP=$CODE，链将依赖 schedule 重燃"
            head -c 300 "$RUNNER_TEMP/relay_resp.txt" 2>/dev/null || true
          fi
          # 接力失败不算 job 失败：蹲守工作已正常完成一轮。
          exit 0
===== END =====
